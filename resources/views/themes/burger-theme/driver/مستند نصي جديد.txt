        function initTheme() {
            if (AppState.currentTheme === 'dark') {
                document.documentElement.classList.add('dark');
                document.getElementById('theme-toggle-dark')?.classList.add('active');
                document.getElementById('theme-toggle-light')?.classList.remove('active');
            } else {
                document.documentElement.classList.remove('dark');
                document.getElementById('theme-toggle-light')?.classList.add('active');
                document.getElementById('theme-toggle-dark')?.classList.remove('active');
            }
        }

        function setTheme(theme) {
            AppState.currentTheme = theme;
            localStorage.setItem('theme', theme);
            initTheme();

            // تحديث بلاط الخريطة ديناميكياً ليناسب الوضع الجديد
            if (AppState.map && AppState.tileLayer) {
                const tileUrl = theme === 'dark' ? TrackingConfig.mapTiles.dark : TrackingConfig.mapTiles.light;
                AppState.tileLayer.setUrl(tileUrl);
            }
        }

            AppState.tileLayer = L.tileLayer(tileUrl, {
                maxZoom: 19,
                subdomains: 'abcd'
            }).addTo(AppState.map);
        }

        /**
         * 4. جلب ورسم المسار المباشر باستخدام خدمة OSRM (Open Source Routing Machine)
         */
        async function fetchAndDrawRoute() {
            const start = TrackingConfig.restaurantCoords;
            const end = TrackingConfig.destinationCoords;
            
            // رابط API لخدمة OSRM لحساب أقصر مسار بري
            const osrmUrl = `https://router.project-osrm.org/route/v1/driving/${start[1]},${start[0]};${end[1]},${end[0]}?overview=full&geometries=geojson`;

            try {
                const response = await fetch(osrmUrl);
                const data = await response.json();

                if (data.routes && data.routes.length > 0) {
                    const routeCoordinates = data.routes[0].geometry.coordinates.map(coord => [coord[1], coord[0]]);

                    // إزالة المسار القديم إن وجد
                    if (AppState.routePolyline) {
                        AppState.map.removeLayer(AppState.routePolyline);
                    }

                    // رسم المسار الجديد كخط متصل ملون
                    AppState.routePolyline = L.polyline(routeCoordinates, {
                        color: '#3B82F6',
                        weight: 5,
                        opacity: 0.8,
                        dashArray: '8, 12',
                        lineCap: 'round'
                    }).addTo(AppState.map);

                    // تحديث زمن الوصول المتبقي والمسافة الأولية
                    updateTripMetrics(data.routes[0].distance, data.routes[0].duration);
                }
            } catch (error) {
                console.warn('تعذر جلب المسار عبر OSRM، سيتم استخدام الخط الافتراضي:', error);
                
                // مسار احتياطي في حال تعذر الاتصال بالخدمة الخارجية
                AppState.routePolyline = L.polyline([start, TrackingConfig.courierInitialCoords, end], {
                    color: '#3B82F6',
                    weight: 4,
                    opacity: 0.7,
                    dashArray: '5, 10'
                }).addTo(AppState.map);
            }
        }

        /**
         * 5. نظام التحريك الانسيابي للعلامة (Smooth Marker Animation Engine)
         */
        let animationFrameId = null;

        function animateCourierMarker(targetLat, targetLng, targetSpeed = 30, duration = 2000) {
            if (!AppState.markers.courier) return;

            const marker = AppState.markers.courier;
            const startLatLng = marker.getLatLng();
            const startLat = startLatLng.lat;
            const startLng = startLatLng.lng;
            
            const startTime = performance.now();

            // حساب زاوية الاتجاه (Bearing / Heading) لتدوير أيقونة الكابتن نحو اتجاه الحركة
            const bearing = calculateBearing(startLat, startLng, targetLat, targetLng);
            rotateCourierIcon(bearing);

            // إلغاء أي حركة سابقة معلقة لتجنب التداخل
            if (animationFrameId) {
                cancelAnimationFrame(animationFrameId);
            }

            function step(currentTime) {
                const elapsedTime = currentTime - startTime;
                const progress = Math.min(elapsedTime / duration, 1);

                // تطبيق التنعيم التكعيبي (Cubic Easing Out) لضمان حركة سلسة وطبيعية
                const easeProgress = 1 - Math.pow(1 - progress, 3);

                const currentLat = startLat + (targetLat - startLat) * easeProgress;
                const currentLng = startLng + (targetLng - startLng) * easeProgress;

                marker.setLatLng([currentLat, currentLng]);

                if (progress < 1) {
                    animationFrameId = requestAnimationFrame(step);
                } else {
                    // تحديث المسافة والزمن المتبقي فور اكتمال نقطة الحركة
                    recalculateRemainingDistance([targetLat, targetLng]);
                }
            }

            animationFrameId = requestAnimationFrame(step);
            
            // تحديث مؤشر السرعة اللحظية في الواجهة
            const speedElement = document.getElementById('stat-current-speed');
            if (speedElement) speedElement.textContent = `${Math.round(targetSpeed)} كم/س`;
        }

        /**
         * حساب زاوية الاتجاه الجغرافي بالدرجات (Bearing)
         */
        function calculateBearing(startLat, startLng, destLat, destLng) {
            const startLatRad = startLat * (Math.PI / 180);
            const destLatRad = destLat * (Math.PI / 180);
            const dLng = (destLng - startLng) * (Math.PI / 180);

            const y = Math.sin(dLng) * Math.cos(destLatRad);
            const x = Math.cos(startLatRad) * Math.sin(destLatRad) -
                      Math.sin(startLatRad) * Math.cos(destLatRad) * Math.cos(dLng);

            let brng = Math.atan2(y, x) * (180 / Math.PI);
            return (brng + 360) % 360;
        }

        /**
         * تطبيق تدوير الأيقونة باستخدام التحويلات البصرية CSS Transforms
         */
        function rotateCourierIcon(deg) {
            const courierElement = document.querySelector('.marker-courier');
            if (courierElement) {
                courierElement.style.transform = `rotate(${deg}deg)`;
                courierElement.style.transition = 'transform 0.5s ease-out';
            }
        }

        /**
         * 6. إعادة حساب المسافة المتبقية والزمن التقديري (Metrics Calculation)
         */
        function recalculateRemainingDistance(currentCourierCoords) {
            const dest = L.latLng(TrackingConfig.destinationCoords[0], TrackingConfig.destinationCoords[1]);
            const courier = L.latLng(currentCourierCoords[0], currentCourierCoords[1]);

            // حساب المسافة المباشرة بالمتر
            const distanceMeters = courier.distanceTo(dest);
            const distanceKm = (distanceMeters / 1000).toFixed(1);

            // افتراض متوسط سرعة 30 كم/س للقيادة داخل المدينة
            const estimatedMinutes = Math.ceil((distanceMeters / 1000) / 30 * 60);

            const distanceElem = document.getElementById('stat-remaining-distance');
            const etaElem = document.getElementById('stat-eta-time');

            if (distanceElem) distanceElem.textContent = `${distanceKm} كم`;
            if (etaElem) etaElem.textContent = `${estimatedMinutes > 0 ? estimatedMinutes : 1} دقائق`;
        }

        function updateTripMetrics(meters, seconds) {
            const distanceKm = (meters / 1000).toFixed(1);
            const minutes = Math.ceil(seconds / 60);

            const distanceElem = document.getElementById('stat-remaining-distance');
            const etaElem = document.getElementById('stat-eta-time');

            if (distanceElem) distanceElem.textContent = `${distanceKm} كم`;
            if (etaElem) etaElem.textContent = `${minutes} دقائق`;
        }
        /**
         * 7. الربط المباشر مع Laravel Echo / WebSockets لاستقبال بيانات البوصلة والموقع
         */
        function initRealtimeTracking() {
            if (typeof Echo === 'undefined') {
                console.warn('Laravel Echo غير معرف في البيئة الحالية، سيتم الانتقال للوضع المحاكي (Demo Mode).');
                runDemoSimulation();
                return;
            }

            // تهيئة اتصال Laravel Echo عبر Pusher
            window.Echo = new Echo({
                broadcaster: 'pusher',
                key: TrackingConfig.pusherKey,
                cluster: TrackingConfig.pusherCluster,
                forceTLS: true
            });

            // الاستماع للقناة الخاصة بهذا الطلب محددة بـ Order ID
            window.Echo.private(`orders.${TrackingConfig.orderId}`)
                .listen('.CourierPositionUpdated', (e) => {
                    /*
                     * البيانات القادمة من البث المباشر (Payload):
                     * e.lat     : خط العرض الحالي
                     * e.lng     : خط الطول الحالي
                     * e.speed   : السرعة الحالية
                     * e.heading : زاوية اتجاه البوصلة الحقيقية من حساس الجوال (Compass Heading)
                     */
                    const { lat, lng, speed, heading } = e;

                    // 1. تحديث الموقع وتحريك العلامة انسيابياً
                    animateCourierMarker(lat, lng, speed, 2000);

                    // 2. تحديث زاوية اتجاه البوصلة الحقيقية إذا كانت متوفرة من الجهاز
                    if (heading !== undefined && heading !== null) {
                        applyCompassHeading(heading);
                    }
                })
                .listen('.OrderStatusUpdated', (e) => {
                    updateOrderStatusUI(e.status, e.timestamp);
                });
        }

        /**
         * 8. معالجة وتطبيق زاوية البوصلة (Compass / Orientation Handler)
         * تقوم هذه الدالة بتدوير أيقونة السائق بدقة فائقة وفق القطب الشمالي أو اتجاه الحركة
         */
        function applyCompassHeading(headingDegree) {
            const courierElement = document.querySelector('.marker-courier');
            if (!courierElement) return;

            // توحيد قيمة الزاوية ضمن النطاق 0 - 360 درجة
            const normalizedDegree = (headingDegree % 360 + 360) % 360;

            // تطبيق التدوير عبر CSS Transform مع انتقال انسيابي لمنع الاهتزاز
            courierElement.style.transform = `rotate(${normalizedDegree}deg)`;
            courierElement.style.transition = 'transform 0.4s cubic-bezier(0.25, 0.1, 0.25, 1)';
        }

        /**
         * 9. نظام المحاكاة والاختبار التجريبي (Demo Mode) لدعم قراءات البوصلة والموقع تلقائياً
         */
        function runDemoSimulation() {
            console.log('جاري بدء وضع المحاكاة اللحظية للبوصلة وحركة الكابتن...');

            // مسار نقاط وهمي يتضمن اختبار زوايا البوصلة (Heading in Degrees)
            const demoWaypoints = [
                { lat: 36.2021, lng: 37.1343, speed: 22, heading: 45 },
                { lat: 36.2050, lng: 37.1375, speed: 35, heading: 68 },
                { lat: 36.2085, lng: 37.1410, speed: 40, heading: 90 },
                { lat: 36.2110, lng: 37.1460, speed: 28, heading: 125 },
                { lat: 36.2150, lng: 37.1520, speed: 12, heading: 160 }
            ];

            let index = 0;

            const demoInterval = setInterval(() => {
                if (index >= demoWaypoints.length) {
                    clearInterval(demoInterval);
                    // تحديث الحالة إلى "تم التوصيل" عند النقطة الأخيرة
                    const currentTime = new Date().toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' });
                    updateOrderStatusUI('delivered', currentTime);
                    return;
                }

                const currentPoint = demoWaypoints[index];

                // تحريك العلامة للموقع الجديد
                animateCourierMarker(currentPoint.lat, currentPoint.lng, currentPoint.speed, 2500);

                // تطبيق زاوية البوصلة للمحاكاة
                applyCompassHeading(currentPoint.heading);

                index++;
            }, 4000);
        }

        /**
         * 10. تحديث واجهة حالة الطلب والمخطط الزمني (Status UI Handler)
         */
        function updateOrderStatusUI(status, timestamp) {
            const badge = document.getElementById('order-status-badge');
            const statusText = document.getElementById('status-text');

            if (!badge || !statusText) return;

            badge.className = `status-pill ${status}`;

            const statusTranslations = {
                'pending': 'قيد الانتظار',
                'preparing': 'جاري التحضير',
                'on_way': 'جاري التوصيل',
                'delivered': 'تم التوصيل'
            };

            statusText.textContent = statusTranslations[status] || status;

            // عند الوصول والتسليم
            if (status === 'delivered') {
                const step3 = document.getElementById('step-node-3');
                const step4 = document.getElementById('step-node-4');
                const timeStep4 = document.getElementById('time-step-4');

                if (step3) step3.classList.replace('active', 'completed');
                if (step4) step4.classList.add('completed', 'active');
                if (timeStep4 && timestamp) timeStep4.textContent = timestamp;

                const progressBar = document.getElementById('stepper-progress');
                if (progressBar) progressBar.style.height = '100%';

                const speedStat = document.getElementById('stat-current-speed');
                if (speedStat) speedStat.textContent = '0 كم/س';
            }
        }

    <script>
        /**
         * 11. وظائف إدارة نافذة الدردشة المباشرة (Chat Modal Logic)
         */
        function openChatModal() {
            const modal = document.getElementById('chat-modal');
            if (modal) {
                modal.classList.remove('hidden');
                document.getElementById('chat-input')?.focus();
            }
        }

        function closeChatModal() {
            const modal = document.getElementById('chat-modal');
            if (modal) {
                modal.classList.add('hidden');
            }
        }

        function handleKeyPress(e) {
            if (e.key === 'Enter') {
                sendMessage();
            }
        }

        function sendMessage() {
            const input = document.getElementById('chat-input');
            const messagesContainer = document.getElementById('chat-messages');

            if (!input || !messagesContainer || !input.value.trim()) return;

            const text = input.value.trim();
            const timeStr = new Date().toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' });

            const msgHtml = `
                <div class="flex justify-end">
                    <div class="bg-primary-600 text-white text-xs p-3 rounded-2xl rounded-tl-none max-w-[80%] leading-relaxed shadow-sm">
                        ${escapeHtml(text)}
                        <span class="block text-[10px] text-blue-200 mt-1">${timeStr}</span>
                    </div>
                </div>
            `;

            messagesContainer.insertAdjacentHTML('beforeend', msgHtml);
            input.value = '';
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            if (window.Echo && TrackingConfig.orderId) {
                fetch(`/api/orders/${TrackingConfig.orderId}/chat`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': TrackingConfig.csrfToken
                    },
                    body: JSON.stringify({ message: text })
                }).catch(() => console.log('وضع المحاكاة: تم إرسال الرسالة للواجهة المحلية.'));
            }
        }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        // إغلاق النافذة عند الضغط على مفتاح Escape
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                closeChatModal();
            }
        });
